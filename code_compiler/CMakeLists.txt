# The licenses of BlitzLLVM and of every bundled dependency that ends up in the program (the ones of the system's libraries are
# shipped by their packages): <licenses>/BlitzLLVM, SDL3, SDL3_image, SDL3_mixer
# Vulkan-Headers, stb_truetype and VulkanMemoryAllocator. SDL_image and SDL_mixer install their own license and those of the codecs they were built with (libwebp,
# libjxl, libavif, dav1d, ...), which are taken from their install directories. The others are taken from their sources.
function(blitz_install_licenses DESTINATION)
	set(_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
	install(FILES "${_ROOT}/LICENSE.adoc" DESTINATION "${DESTINATION}/BlitzLLVM" ${ARGN})
	if(BLITZRT_BUNDLED_SDL3)
		install(FILES "${_ROOT}/third_party/SDL/LICENSE.txt" DESTINATION "${DESTINATION}/SDL3" ${ARGN})
	endif()
	install(DIRECTORY "${_ROOT}/third_party/Vulkan-Headers/LICENSES" DESTINATION "${DESTINATION}/Vulkan-Headers" ${ARGN})
	install(FILES "${_ROOT}/third_party/Vulkan-Headers/LICENSE.md" DESTINATION "${DESTINATION}/Vulkan-Headers" ${ARGN})
	install(FILES "${_ROOT}/third_party/stb/LICENSE" DESTINATION "${DESTINATION}/stb" ${ARGN})
	install(FILES "${_ROOT}/third_party/VulkanMemoryAllocator/LICENSE.txt" DESTINATION "${DESTINATION}/VulkanMemoryAllocator" ${ARGN})
	foreach(_DIRECTORY IN LISTS BLITZRT_LICENSES)
		install(DIRECTORY "${_DIRECTORY}/" DESTINATION "${DESTINATION}" ${ARGN})
	endforeach()
endfunction()
blitz_install_licenses(licenses)

# Tests: the function test suite and the command line options, each below. The lexer test only has to lex.
enable_testing()
set(TEST_DIRECTORY "${PROJECT_SOURCE_DIR}/../tests")
# The function test suite: a test for every function of Blitz3D and BlitzPlus, in tests/suite. Each file is a program
# that stops with an error when something is wrong. Files in pending/ are for what is not implemented yet and are
# expected not to compile, files in fail/ are expected to compile but to stop with an error (the one that a line "; Expect: <regex>" names, if there is one), files in reject/ are expected to be refused by the compiler. Programs that need a
# window and a GPU end with _pixels, and only run for real with -DBLITZ_TEST_GPU=ON.
option(BLITZ_TEST_GPU "Also run the tests that need a window and a GPU" OFF)
file(MAKE_DIRECTORY "${PROJECT_BINARY_DIR}/suite")
# The Draw3D2 tests (tests/suite/draw3d2_*) use the Draw3D2 userlib of tests/draw3d2, which calls two functions of user32.dll. Where there is no
# user32.dll a small library of the same name next to the programs stands in for it.
set(_DRAW3D2_USERLIBS --userlibs-dir "${TEST_DIRECTORY}/draw3d2/userlibs")
if(NOT WIN32)
	add_library(draw3d2_user32 SHARED "${TEST_DIRECTORY}/draw3d2/shim/user32_shim.c")
	set_target_properties(draw3d2_user32 PROPERTIES OUTPUT_NAME user32 LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/suite")
endif()
file(GLOB_RECURSE _SUITE_TESTS CONFIGURE_DEPENDS "${TEST_DIRECTORY}/suite/*.bb")
foreach(_FILE IN LISTS _SUITE_TESTS)
	file(RELATIVE_PATH _RELATIVE "${TEST_DIRECTORY}/suite" "${_FILE}")
	if(_RELATIVE STREQUAL "check.bb")
		continue()
	endif()
	string(REGEX REPLACE "\\.bb$" "" _NAME "${_RELATIVE}")
	string(REPLACE "/" "-" _NAME "${_NAME}")
	# Every program is built twice: as it is by default (Release), and with -d (Debug: debug info, and -d implies --profile).
	# The Debug tests are named suite-debug-compile-* and suite-debug-run-*, the Release ones keep their names.
	foreach(_CONFIG IN ITEMS release debug)
		if(_CONFIG STREQUAL "debug")
			set(_OPTIONS -d)
			set(_PREFIX "suite-debug")
			set(_SUFFIX "-debug")
		else()
			set(_OPTIONS)
			set(_PREFIX "suite")
			set(_SUFFIX)
		endif()
		set(_EXE "${PROJECT_BINARY_DIR}/suite/${_NAME}${_SUFFIX}${CMAKE_EXECUTABLE_SUFFIX}")
		if(_NAME MATCHES "^draw3d2_")
			list(APPEND _OPTIONS ${_DRAW3D2_USERLIBS})
		endif()
		if(_RELATIVE MATCHES "^reject/userlib_")
			list(APPEND _OPTIONS --userlibs-dir "${TEST_DIRECTORY}/cli/userlibs") # CliAdd% and the others of cli/userlibs/test.decls.
		endif()
		if(_RELATIVE MATCHES "^reject/")
			# Programs that are wrong and have to be refused by the compiler.
			add_test(NAME "${_PREFIX}-compile-${_NAME}" COMMAND ${PROJECT_NAME} ${_OPTIONS} "${_FILE}" -o "${_EXE}")
			# A line "; Expect: <regex>" names the error it has to be refused with.
			file(STRINGS "${_FILE}" _EXPECT REGEX "^; Expect: " LIMIT_COUNT 1)
			if(_EXPECT)
				string(REGEX REPLACE "^.*Expect: " "" _EXPECT "${_EXPECT}")
			else()
				set(_EXPECT "Type mismatch|Division by zero")
			endif()
			set_tests_properties("${_PREFIX}-compile-${_NAME}" PROPERTIES PASS_REGULAR_EXPRESSION "${_EXPECT}" LABELS "suite")
			continue()
		endif()
		if(_RELATIVE MATCHES "^fail/.*_pixels\\.bb$" AND NOT BLITZ_TEST_GPU)
			continue()  # Needs a window and a GPU to fail the way it is meant to.
		endif()
		if(_RELATIVE MATCHES "^pending/" AND _CONFIG STREQUAL "debug")
			continue()  # A placeholder is the same in both, so it is only checked once.
		endif()
		add_test(NAME "${_PREFIX}-compile-${_NAME}" COMMAND ${PROJECT_NAME} ${_OPTIONS} "${_FILE}" -o "${_EXE}")
		if(_RELATIVE MATCHES "^pending/")
			set_tests_properties("${_PREFIX}-compile-${_NAME}" PROPERTIES WILL_FAIL TRUE LABELS "pending")
			continue()
		endif()
		set_tests_properties("${_PREFIX}-compile-${_NAME}" PROPERTIES FIXTURES_SETUP "${_PREFIX}-${_NAME}" LABELS "suite")
		add_test(NAME "${_PREFIX}-run-${_NAME}" COMMAND "${_EXE}" WORKING_DIRECTORY "${TEST_DIRECTORY}/suite")
		set_tests_properties("${_PREFIX}-run-${_NAME}" PROPERTIES FIXTURES_REQUIRED "${_PREFIX}-${_NAME}" LABELS "suite")
		if(_NAME STREQUAL "blitz3d_many_lights_pixels")
			# Makes the runtime check the lights that the cells of the view pick against those of all lights.
			set_property(TEST "${_PREFIX}-run-${_NAME}" APPEND PROPERTY ENVIRONMENT "BLITZLLVM_DEBUG_CHECK_LIGHTS=1")
		endif()
		if(_NAME MATCHES "_pixels$" AND BLITZ_TEST_GPU)
			# The program is told where it is, for the ones that start themselves again (graphics_save_async_pixels).
			set_tests_properties("${_PREFIX}-run-${_NAME}" PROPERTIES LABELS "suite;gpu" ENVIRONMENT "BLITZLLVM_TEST_PROGRAM=${_EXE}")
		else()
			set_tests_properties("${_PREFIX}-run-${_NAME}" PROPERTIES ENVIRONMENT "BLITZLLVM_HEADLESS=1;BLITZLLVM_TEST_PROGRAM=${_EXE}")
		endif()
		# The Release and the Debug run of a program use the same files in tests/suite (suite_tmp_*) and ports, so they take turns.
		set_property(TEST "${_PREFIX}-run-${_NAME}" APPEND PROPERTY RESOURCE_LOCK "suite-${_NAME}")
		if((_NAME STREQUAL "blitz3d_dither_blend_pixels" OR _NAME STREQUAL "blitz3d_dither_blend_deep_pixels") AND BLITZ_TEST_GPU)
			# Again without the local read extensions, which is how the pixel interlock gets used where it is there.
			add_test(NAME "${_PREFIX}-run-${_NAME}-interlock" COMMAND "${_EXE}" WORKING_DIRECTORY "${TEST_DIRECTORY}/suite")
			set_tests_properties("${_PREFIX}-run-${_NAME}-interlock" PROPERTIES FIXTURES_REQUIRED "${_PREFIX}-${_NAME}" LABELS "suite;gpu" ENVIRONMENT "BLITZLLVM_TEST_PROGRAM=${_EXE};BLITZLLVM_DITHER_LOCAL_READ=0" RESOURCE_LOCK "suite-${_NAME}")
		endif()
		if(_NAME STREQUAL "network" OR _NAME STREQUAL "multiplayer")
			set_property(TEST "${_PREFIX}-run-${_NAME}" APPEND PROPERTY RESOURCE_LOCK "suite-network-ports")
		endif()
		if(_CONFIG STREQUAL "debug")
			# For the few programs that behave differently in a debug build (Stop is a breakpoint there).
			set_property(TEST "${_PREFIX}-run-${_NAME}" APPEND PROPERTY ENVIRONMENT "BLITZLLVM_TEST_DEBUG=1")
		endif()
		if(_RELATIVE MATCHES "^fail/")
			# A program can say which error it has to stop with, in a line that starts with "; Expect: ", which is a regular expression. Any other
			# way of failing (a crash included) is then not good enough.
			file(STRINGS "${_FILE}" _EXPECT REGEX "^; Expect: " LIMIT_COUNT 1)
			if(_EXPECT)
				string(REGEX REPLACE "^.*Expect: " "" _EXPECT "${_EXPECT}") # file(STRINGS) escapes the ; of the comment
				set_tests_properties("${_PREFIX}-run-${_NAME}" PROPERTIES PASS_REGULAR_EXPRESSION "${_EXPECT}")
			else()
				set_tests_properties("${_PREFIX}-run-${_NAME}" PROPERTIES WILL_FAIL TRUE)
			endif()
		endif()
	endforeach()
endforeach()

		if("${_PROGRAM}" IN_LIST _ORIGINAL_RUN_KNOWN)

		endif()
	endforeach()
endif()
