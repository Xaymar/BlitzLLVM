# What 'blitzcc -v' reports. PROJECT_VERSION is reset by project() above, so take the one of the top level project.
target_compile_definitions(${PROJECT_NAME} PRIVATE BLITZLLVM_VERSION="${CMAKE_PROJECT_VERSION}")

# The runtime library has to live next to the compiler, which links it into every program it produces.
add_dependencies(${PROJECT_NAME} blitzrt)
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${PROJECT_NAME}>/runtime"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:blitzrt>" ${BLITZRT_LIBRARIES} "$<TARGET_FILE_DIR:${PROJECT_NAME}>/runtime/"
)

# Install layout: <prefix>/bin/blitzcc, <prefix>/bin/runtime and <prefix>/userlibs.
install(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION bin)
install(FILES "$<TARGET_FILE:blitzrt>" ${BLITZRT_LIBRARIES} DESTINATION bin/runtime)
install(FILES "userlibs/README.txt" DESTINATION userlibs)
install(DIRECTORY "${BLITZRT_LICENSES}/" DESTINATION licenses)

# System layout for packages (deb, rpm, pkg), left out of a plain install: <prefix>/bin/blitzcc, <prefix>/lib/blitzllvm/runtime
# and <prefix>/share/blitzllvm/userlibs. Only the packaging step installs this component, see the end of the top level CMakeLists.txt.
install(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION bin COMPONENT system EXCLUDE_FROM_ALL)
install(FILES "$<TARGET_FILE:blitzrt>" ${BLITZRT_LIBRARIES} DESTINATION lib/blitzllvm/runtime COMPONENT system EXCLUDE_FROM_ALL)
install(FILES "userlibs/README.txt" DESTINATION share/blitzllvm/userlibs COMPONENT system EXCLUDE_FROM_ALL)
install(DIRECTORY "${BLITZRT_LICENSES}/" DESTINATION share/licenses/blitzllvm COMPONENT system EXCLUDE_FROM_ALL)

# Tests: every program in /tests has to compile into an executable, the lexer test only has to lex.
enable_testing()
set(TEST_DIRECTORY "${PROJECT_SOURCE_DIR}/../tests")
foreach(_TEST IN ITEMS 001 002 003 004 005 006 007 008 009 010 011 012 013 014 015 016 017 018 019 020 021 022 023 024)
	add_test(
		NAME "compile-${_TEST}"
		COMMAND ${PROJECT_NAME} "${TEST_DIRECTORY}/${_TEST}.bb" -o "${PROJECT_BINARY_DIR}/test-${_TEST}${CMAKE_EXECUTABLE_SUFFIX}"
	)
	set_tests_properties("compile-${_TEST}" PROPERTIES FIXTURES_SETUP "program-${_TEST}")
	# Run the programs too. Without a window, so that this also works on machines without a display or GPU.
	add_test(
		NAME "run-${_TEST}"
		COMMAND "${PROJECT_BINARY_DIR}/test-${_TEST}${CMAKE_EXECUTABLE_SUFFIX}"
	)
	set_tests_properties("run-${_TEST}" PROPERTIES FIXTURES_REQUIRED "program-${_TEST}" ENVIRONMENT "BLITZ_HEADLESS=1" WORKING_DIRECTORY "${TEST_DIRECTORY}")
endforeach()
# The function test suite: a test for every function of Blitz3D and BlitzPlus, in tests/suite. Each file is a program
# that stops with an error when something is wrong. Files in pending/ are for what is not implemented yet and are
# expected not to compile, files in fail/ are expected to compile but to stop with an error. Programs that need a
# window and a GPU end with _pixels, and only run for real with -DBLITZ_TEST_GPU=ON.
option(BLITZ_TEST_GPU "Also run the tests that need a window and a GPU" OFF)
file(MAKE_DIRECTORY "${PROJECT_BINARY_DIR}/suite")
file(GLOB_RECURSE _SUITE_TESTS CONFIGURE_DEPENDS "${TEST_DIRECTORY}/suite/*.bb")
foreach(_FILE IN LISTS _SUITE_TESTS)
	file(RELATIVE_PATH _RELATIVE "${TEST_DIRECTORY}/suite" "${_FILE}")
	if(_RELATIVE STREQUAL "check.bb")
		continue()
	endif()
	string(REGEX REPLACE "\.bb$" "" _NAME "${_RELATIVE}")
	string(REPLACE "/" "-" _NAME "${_NAME}")
	set(_EXE "${PROJECT_BINARY_DIR}/suite/${_NAME}${CMAKE_EXECUTABLE_SUFFIX}")
	add_test(NAME "suite-compile-${_NAME}" COMMAND ${PROJECT_NAME} "${_FILE}" -o "${_EXE}")
	if(_RELATIVE MATCHES "^pending/")
		set_tests_properties("suite-compile-${_NAME}" PROPERTIES WILL_FAIL TRUE LABELS "pending")
		continue()
	endif()
	set_tests_properties("suite-compile-${_NAME}" PROPERTIES FIXTURES_SETUP "suite-${_NAME}" LABELS "suite")
	add_test(NAME "suite-run-${_NAME}" COMMAND "${_EXE}" WORKING_DIRECTORY "${TEST_DIRECTORY}/suite")
	set_tests_properties("suite-run-${_NAME}" PROPERTIES FIXTURES_REQUIRED "suite-${_NAME}" LABELS "suite")
	if(_NAME MATCHES "_pixels$" AND BLITZ_TEST_GPU)
		set_tests_properties("suite-run-${_NAME}" PROPERTIES LABELS "suite;gpu")
	else()
		set_tests_properties("suite-run-${_NAME}" PROPERTIES ENVIRONMENT "BLITZ_HEADLESS=1;BLITZ_SUITE_PROGRAM=${_EXE}")
	endif()
	if(_RELATIVE MATCHES "^fail/")
		set_tests_properties("suite-run-${_NAME}" PROPERTIES WILL_FAIL TRUE)
	endif()
endforeach()

# The command line options of the compiler: -q, +q, -c, -d, -k, +k, -v. Each case is in tests/cli/cli.cmake.
foreach(_CASE IN ITEMS default quiet very_quiet options_not_case_sensitive compile_only debug keywords keywords_syntax keywords_bad_userlib version verbose_long_only help)
	add_test(
		NAME "cli-${_CASE}"
		COMMAND ${CMAKE_COMMAND}
			"-DCASE=${_CASE}"
			"-DBLITZCC=$<TARGET_FILE:${PROJECT_NAME}>"
			"-DTESTS=${TEST_DIRECTORY}"
			"-DWORK=${PROJECT_BINARY_DIR}/cli/${_CASE}"
			"-DVERSION=${CMAKE_PROJECT_VERSION}"
			-P "${TEST_DIRECTORY}/cli/cli.cmake"
	)
	set_tests_properties("cli-${_CASE}" PROPERTIES LABELS "cli")
endforeach()
