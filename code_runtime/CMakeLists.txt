################################################################################
# Runtime Library
#
# Linked into every program that the compiler produces.
################################################################################
add_library(blitzrt STATIC)
init_project(blitzrt)

# The allocator (memory.cpp) always replaces operator new/delete. On Linux with glibc it also replaces malloc and free (memory_interpose.c),
# which puts the libraries a program loads on it too; elsewhere SDL is pointed at it with SDL_SetMemoryFunctions. The runtime's own sources call
# bb_malloc, bb_free and so on directly.
include(CheckSymbolExists)
set(_REPLACE_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	check_symbol_exists(__GLIBC__ "features.h" BLITZ_HAVE_GLIBC)
	set(_REPLACE_DEFAULT ${BLITZ_HAVE_GLIBC})
endif()
option(BLITZ_REPLACE_MALLOC "Replace malloc and free of programs with the runtime's allocator (glibc only)" ${_REPLACE_DEFAULT})
option(BLITZ_CUSTOM_ALLOCATOR "Use the runtime's own allocator at all (OFF is for comparing it with the system's)" ON)

if(NOT BLITZ_CUSTOM_ALLOCATOR)
	# The functions exist (the sources of programs call them) but only pass on to the system; new and delete are not replaced.
	target_compile_definitions(blitzrt PRIVATE BB_ALLOC_PASSTHROUGH)
elseif(BLITZ_REPLACE_MALLOC)
	target_sources(blitzrt PRIVATE "source/memory_interpose.c")
	target_compile_definitions(blitzrt PRIVATE BB_INTERPOSE_MALLOC)
endif()
target_include_directories(blitzrt PRIVATE
	"${CMAKE_CURRENT_SOURCE_DIR}/source"
)
target_include_directories(blitzrt SYSTEM PRIVATE
	${BLITZRT_INCLUDE_DIRECTORIES}
	"${PROJECT_SOURCE_DIR}/third_party/stb"
	"${PROJECT_SOURCE_DIR}/third_party/VulkanMemoryAllocator/include"
)
if(NOT MSVC AND NOT APPLE)
	# See dependencies.cmake: programs are linked with --gc-sections, which only drops what is in a section of its own.
	target_compile_options(blitzrt PRIVATE -ffunction-sections -fdata-sections)
endif()
if(NOT MSVC)
	# Float results have to be the same on every CPU (they are compared across the network): no fused multiply-add where the CPU has one,
	# which GCC does by default and clang within an expression. MSVC does not contract with /fp:precise, its default.
	target_compile_options(blitzrt PRIVATE -ffp-contract=off)
endif()
if(MSVC)
	# No C++ runtime dependencies: the library is linked by clang into programs.
	target_compile_options(blitzrt PRIVATE /GR- /EHs-c- /Zc:preprocessor)
	target_compile_definitions(blitzrt PRIVATE _HAS_EXCEPTIONS=0)
	# The vectorized STL algorithms need helpers (__std_rotate, ...) that older toolsets' msvcprt.lib lacks, see dependencies.cmake.
	target_compile_definitions(blitzrt PRIVATE _USE_STD_VECTOR_ALGORITHMS=0 _USE_STD_VECTOR_FLOATING_ALGORITHMS=0)
	# Programs are linked by clang, which uses the dynamic release C runtime.
	target_compile_options(blitzrt PRIVATE /MD)
endif()

get_target_property(_SOURCES blitzrt SOURCES)
source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${_SOURCES})
