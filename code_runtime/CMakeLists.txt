# bb_malloc, bb_free and so on directly.

	check_symbol_exists(__GLIBC__ "features.h" BLITZ_HAVE_GLIBC)
	set(_REPLACE_DEFAULT ${BLITZ_HAVE_GLIBC})
endif()

option(BLITZ_CUSTOM_ALLOCATOR "Use the runtime's own allocator at all (OFF is for comparing it with the system's)" ON)

# The shaders of the Direct3D 11 backend. source/d3d11 holds the HLSL that tools/translate_shaders.py makes from the SPIR-V (CI checks that
# it is current). Direct3D 11 takes bytecode, and the program must not need the shader compiler at run time, so the HLSL is compiled here
# with fxc, the compiler of the Windows SDK, into one header per shader (an array named after the shader). Without fxc the runtime is built
# without them, and so without that backend.
option(BLITZLLVM_REQUIRE_FXC "Fail instead of building without the Direct3D 11 shaders when fxc is missing on Windows" OFF)
# The OpenGL backend (BLITZLLVM_RENDERER=opengl), on Linux and Windows. macOS has no OpenGL 4.3. The backend gets every function of
# OpenGL from SDL when it starts, so the runtime and the programs do not link to an OpenGL library.
if(WIN32 OR (UNIX AND NOT APPLE))
	target_sources(blitzrt PRIVATE "source/graphics_opengl.cpp" "source/shaders_opengl.inc")
	target_compile_definitions(blitzrt PRIVATE BB_OPENGL)
endif()

if(WIN32)
	set(_PROGRAM_FILES_X86 "ProgramFiles(x86)")
	file(GLOB _KITS LIST_DIRECTORIES true "$ENV{${_PROGRAM_FILES_X86}}/Windows Kits/10/bin/10.*")
	list(SORT _KITS COMPARE NATURAL ORDER DESCENDING)
	set(_FXC_HINTS "")
	# The compiler for the machine that builds comes first. Its output does not depend on which one runs.
	if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64")
		set(_FXC_ARCHITECTURES arm64 x64 x86)
	else()
		set(_FXC_ARCHITECTURES x64 x86)
	endif()
	foreach(_ARCHITECTURE IN LISTS _FXC_ARCHITECTURES)
		foreach(_KIT IN LISTS _KITS)
			list(APPEND _FXC_HINTS "${_KIT}/${_ARCHITECTURE}")
		endforeach()
	endforeach()
	find_program(BLITZLLVM_FXC NAMES fxc fxc.exe HINTS ${_FXC_HINTS} DOC "fxc, the HLSL compiler of the Windows SDK")
	if(BLITZLLVM_FXC)
		message(STATUS "Direct3D 11 shaders: compiled with ${BLITZLLVM_FXC}")
		file(GLOB _HLSL CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/source/d3d11/*.hlsl")
		list(SORT _HLSL)
		set(_BYTECODE_DIR "${CMAKE_CURRENT_BINARY_DIR}/d3d11")
		file(MAKE_DIRECTORY "${_BYTECODE_DIR}")
		set(_BYTECODE "")
		set(_INCLUDES "// Generated by code_runtime/CMakeLists.txt: the bytecode of every shader of the Direct3D 11 backend.\n")
		foreach(_FILE IN LISTS _HLSL)
			get_filename_component(_NAME "${_FILE}" NAME_WE)
			if(_NAME MATCHES "compute")
				set(_PROFILE cs_5_0)
			elseif(_NAME MATCHES "fragment")
				set(_PROFILE ps_5_0)
			else()
				set(_PROFILE vs_5_0)
			endif()
			file(TO_NATIVE_PATH "${_FILE}" _FILE_NATIVE)
			file(TO_NATIVE_PATH "${_BYTECODE_DIR}/${_NAME}.h" _HEADER_NATIVE)
			add_custom_command(
				OUTPUT "${_BYTECODE_DIR}/${_NAME}.h"
				COMMAND "${BLITZLLVM_FXC}" -nologo -T ${_PROFILE} -E main -O3 -Vn ${_NAME}_dxbc -Fh "${_HEADER_NATIVE}" "${_FILE_NATIVE}"
				DEPENDS "${_FILE}"
				COMMENT "Compiling ${_NAME}.hlsl (${_PROFILE})"
				VERBATIM
			)
			list(APPEND _BYTECODE "${_BYTECODE_DIR}/${_NAME}.h")
			string(APPEND _INCLUDES "#include \"${_NAME}.h\"\n")
		endforeach()
		file(GENERATE OUTPUT "${_BYTECODE_DIR}/shaders_d3d11.inc" CONTENT "${_INCLUDES}")
		# The table that the backend finds its shaders in: SPIR-V size and hash, bytecode, registers (tools/generate_d3d11_table.py).
		find_program(BLITZLLVM_PYTHON NAMES python3 python REQUIRED DOC "Python, which makes the table of the Direct3D 11 shaders")
		file(GLOB _D3D11_MAPS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/source/d3d11/*.d3d11.json")
		add_custom_command(
			OUTPUT "${_BYTECODE_DIR}/shaders_d3d11_table.inc"
			COMMAND "${BLITZLLVM_PYTHON}" "${PROJECT_SOURCE_DIR}/tools/generate_d3d11_table.py" "${_BYTECODE_DIR}/shaders_d3d11_table.inc" "${CMAKE_CURRENT_LIST_DIR}/source"
			DEPENDS "${PROJECT_SOURCE_DIR}/tools/generate_d3d11_table.py" "${CMAKE_CURRENT_LIST_DIR}/source/shaders.inc" "${CMAKE_CURRENT_LIST_DIR}/source/smaa.inc" ${_D3D11_MAPS}
			COMMENT "Making the table of the Direct3D 11 shaders"
			VERBATIM
		)
		list(APPEND _BYTECODE "${_BYTECODE_DIR}/shaders_d3d11_table.inc")
		target_sources(blitzrt PRIVATE "source/graphics_d3d11.cpp")
		add_custom_target(blitzrt_d3d11_shaders DEPENDS ${_BYTECODE})
		add_dependencies(blitzrt blitzrt_d3d11_shaders)
		target_include_directories(blitzrt PRIVATE "${_BYTECODE_DIR}")
		target_compile_definitions(blitzrt PRIVATE BB_D3D11_SHADERS)
	elseif(BLITZLLVM_REQUIRE_FXC)
		message(FATAL_ERROR "fxc was not found, but BLITZLLVM_REQUIRE_FXC is set. It comes with the Windows SDK; set BLITZLLVM_FXC to its path.")
	else()
		message(STATUS "Direct3D 11 shaders: fxc (Windows SDK) not found, the runtime is built without them. Set BLITZLLVM_FXC to its path.")
	endif()
endif()
# What SystemProperty$("runtime") reports: the version and the short commit. The commit is left out without git or a repository.
find_program(GIT NAMES git git.exe)
set(BLITZRT_COMMIT "")
if(GIT AND EXISTS "${GIT}")
	execute_process(
		COMMAND "${GIT}" rev-parse --short=7 HEAD
		WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
		OUTPUT_VARIABLE BLITZRT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
		RESULT_VARIABLE _RESULT ERROR_QUIET
	)
	if(NOT _RESULT EQUAL 0)
		set(BLITZRT_COMMIT "")
	endif()
endif()

	"BLITZLLVM_VERSION=\"${CMAKE_PROJECT_VERSION}\";BLITZLLVM_COMMIT=\"${BLITZRT_COMMIT}\"")

endif()

if(NOT MSVC AND NOT APPLE)
	# See dependencies.cmake: programs are linked with --gc-sections, which only drops what is in a section of its own.
	target_compile_options(blitzrt PRIVATE -ffunction-sections -fdata-sections)
endif()
if(WIN32 AND NOT MSVC AND NOT MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
	# clang uses the STL of Visual Studio on Windows, see dependencies.cmake.
	target_compile_definitions(blitzrt PRIVATE _USE_STD_VECTOR_ALGORITHMS=0 _USE_STD_VECTOR_FLOATING_ALGORITHMS=0)
endif()
if(NOT MSVC)
	# Float results have to be the same on every CPU (they are compared across the network): no fused multiply-add where the CPU has one,
	# which GCC does by default and clang within an expression. MSVC does not contract with /fp:precise, its default.
	target_compile_options(blitzrt PRIVATE -ffp-contract=off)
endif()
if(MSVC AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
	# clang-cl (the Windows build uses it, see build.yml) contracts like clang does, and /fp:precise does not turn that off.
	target_compile_options(blitzrt PRIVATE /clang:-ffp-contract=off)
endif()
if(MSVC)
	# No C++ runtime dependencies: the library is linked by clang into programs.
	target_compile_options(blitzrt PRIVATE /GR- /EHs-c-)
	if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
		target_compile_options(blitzrt PRIVATE /Zc:preprocessor)
	endif()
	target_compile_definitions(blitzrt PRIVATE _HAS_EXCEPTIONS=0)
	# The vectorized STL algorithms need helpers (__std_rotate, ...) that older toolsets' msvcprt.lib lacks, see dependencies.cmake.
	target_compile_definitions(blitzrt PRIVATE _USE_STD_VECTOR_ALGORITHMS=0 _USE_STD_VECTOR_FLOATING_ALGORITHMS=0)
	# Programs are linked by clang, which uses the dynamic release C runtime.
	target_compile_options(blitzrt PRIVATE /MD)
endif()

if(MINGW)
	# Same as the clang-cl build above: no exceptions and no RTTI in the runtime, which the programs link and which has no use for either.
	target_compile_options(blitzrt PRIVATE -fno-exceptions -fno-rtti)
endif()
