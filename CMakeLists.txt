function(init_project TARGET)
	set_target_properties(${TARGET} PROPERTIES
		# Always generate position independent code.
		POSITION_INDEPENDENT_CODE ON

		# Set C++ Standard and Extensions
		C_STANDARD 17
		C_STANDARD_REQUIRED ON
		CXX_STANDARD 20
		CXX_STANDARD_REQUIRED ON
		CXX_EXTENSIONS OFF

		# Remove prefix from generated files.
		PREFIX ""
		IMPORT_PREFIX ""

		# Never treat warnings as errors.
		COMPILE_WARNING_AS_ERROR OFF
	)
	target_compile_definitions(${TARGET} PRIVATE
		__STDC_WANT_LIB_EXT1__=1
	)

	if(WIN32)
		target_compile_definitions(${TARGET} PRIVATE
			# windows.h
			#- Disable MIN/MAX macro as this breaks a lot of code.
			NOMINMAX
			#- Disable IN/OUT macro as this breaks a lot of code.
			NOINOUT
		)
	endif()

	if(MSVC) # Microsoft Visual C/C++
		target_compile_definitions(${TARGET} PRIVATE
			# Disable useless/terrible behavior from MSVC
			_CRT_SECURE_NO_WARNINGS
			_ENABLE_EXTENDED_ALIGNED_STORAGE
		)
		target_compile_options(${TARGET} PRIVATE
			# Dynamically link Microsoft C/C++ Redistributable.
			$<$<CONFIG:>:/MD>
			$<$<CONFIG:Debug>:/MDd>
			$<$<CONFIG:Release>:/MD>
			$<$<CONFIG:RelWithDebInfo>:/MD>
			$<$<CONFIG:MinSizeRel>:/MD>

			# Always compile using multiple processors. Why is this defaulting off anyway?!
			/MP

			# We want full exception support all the time, not conditionally.
			/EHa
		)
	endif()
endfunction()

# The HTML help, in the layout of the Blitz3D help: <prefix>/help in a plain install (the zips), <prefix>/share/doc/blitzllvm/help in the
# packages. It is generated from the wiki text by tools/wiki/wiki.py (and never committed), which needs Python.
option(BLITZLLVM_REQUIRE_HELP "Fail instead of leaving the help out when Python is missing" OFF)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_FOUND)
	add_custom_target(blitz_help ALL
		COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/wiki/wiki.py" help --out "${CMAKE_CURRENT_BINARY_DIR}/help"
		COMMENT "Generating the HTML help"
		VERBATIM)
	install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/help/" DESTINATION help)
	install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/help/" DESTINATION share/doc/blitzllvm/help COMPONENT system EXCLUDE_FROM_ALL)
elseif(BLITZLLVM_REQUIRE_HELP)
	message(FATAL_ERROR "Python was not found, but BLITZLLVM_REQUIRE_HELP is set, and the help is generated with it.")
else()
	message(WARNING "Python was not found, so the HTML help is not built and not installed.")
endif()

# Fail the install when a license is missing, so that rules which silently install nothing (a stale copy of the install
# rules in code_compiler, say) can not ship a bundle without them. Declared after the sub-projects, so it runs after their installs.
set(BLITZ_REQUIRED_LICENSES
	BlitzLLVM/LICENSE.adoc SDL3/LICENSE.txt SDL3_image/LICENSE.txt SDL3_image/optional/LICENSE.dav1d.txt SDL3_image/optional/LICENSE.avif.txt
	SDL3_image/optional/LICENSE.jxl.txt SDL3_image/optional/LICENSE.webp.txt SDL3_mixer/LICENSE.txt SDL3_mixer/optional/LICENSE.opus.txt
	SDL3_mixer/optional/LICENSE.opusfile.txt SDL3_mixer/optional/LICENSE.ogg-vorbis.txt Vulkan-Headers/LICENSE.md
	VulkanMemoryAllocator/LICENSE.txt
)
install(CODE "
	foreach(_LICENSE IN ITEMS ${BLITZ_REQUIRED_LICENSES})
		if(NOT EXISTS \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/licenses/\${_LICENSE}\")
			message(FATAL_ERROR \"The license '\${_LICENSE}' was not installed. Delete the build directory's ExternalContentFiles directory and configure again, or check the license install rules in code_compiler/CMakeLists.txt.\")
		endif()
	endforeach()
")
install(CODE "
	foreach(_LICENSE IN ITEMS ${BLITZ_REQUIRED_LICENSES})
		if(NOT EXISTS \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/share/licenses/blitzllvm/\${_LICENSE}\")
			message(FATAL_ERROR \"The license '\${_LICENSE}' was not installed. Delete the build directory's ExternalContentFiles directory and configure again, or check the license install rules in code_compiler/CMakeLists.txt.\")
		endif()
	endforeach()
" COMPONENT system EXCLUDE_FROM_ALL)
